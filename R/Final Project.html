hist(team_results$PASE)
avgTOP2 <- function(x) {
    sum <- sum(team_results$TOP2[team_results$CHAMP == x])
    n <- length(team_results$TOP2[team_results$CHAMP == x])
    avg = sum/n
    return(avg)
}

avg_TOP2 <- c(avgTOP2(0), avgTOP2(1), avgTOP2(2), avgTOP2(3))

avg_TOP2_df <- data.frame(
CHAMP = c(0:3),
avg_TOP2
)
ggplot(avg_TOP2_df, aes (x = CHAMP, y = avg_TOP2)) +
